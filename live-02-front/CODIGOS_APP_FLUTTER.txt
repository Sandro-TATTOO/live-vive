```
class StreamModel {
  final String id;
  final String title;
  final String thumbnailUrl;
  final int viewerCount;

  StreamModel({
    required this.id,
    required this.title,
    required this.thumbnailUrl,
    required this.viewerCount,
  });

  factory StreamModel.fromJson(Map<String, dynamic> json) {
    return StreamModel(
      id: json['id']?.toString() ?? '',
      title: json['title'] ?? 'Live sem título',
      thumbnailUrl: json['thumbnail_url'] ?? '',
      viewerCount: json['viewer_count'] ?? 0,
    );
  }
}
```

## 4. lib/services/stream_service.dart (API Backend)
```
import 'dart:convert';
import 'package:http/http.dart' as http;
import '../models/stream_model.dart';

class StreamService {
  static const String _baseUrl = 'http://127.0.0.1:8000';  // Backend local

  static Future<List<StreamModel>> fetchStreams() async {
    try {
      final response = await http.get(
        Uri.parse('$_baseUrl/streams/live'),
        headers: {'Content-Type': 'application/json'},
      );

      if (response.statusCode == 200) {
        final List<dynamic> data = json.decode(response.body);
        return data.map((json) => StreamModel.fromJson(json)).toList();
      } else {
        throw Exception('Erro lives: ${response.statusCode}');
      }
    } catch (e) {
      print('❌ StreamService: $e');
      // Mock para testes
      return [
        StreamModel(
          id: '1',
          title: 'Live Teste 1',
          thumbnailUrl: 'https://via.placeholder.com/400x225/FF0000/FFFFFF?text=LIVE+1',
          viewerCount: 123,
        ),
        StreamModel(
          id: '2',
          title: 'Live Teste 2',
          thumbnailUrl: 'https://via.placeholder.com/400x225/00FF00/FFFFFF?text=LIVE+2',
          viewerCount: 456,
        ),
      ];
    }
  }
}
```

## 6. lib/screens/home_screen.dart (Grid Lives)
```
import 'package:flutter/material.dart';
import '../services/stream_service.dart';
import '../models/stream_model.dart';
import '../widgets/live_card.dart';
import 'live_room_screen.dart';  // ← Import

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Lives Ao Vivo', style: TextStyle(fontWeight: FontWeight.bold, fontSize: 20)),
        backgroundColor: Colors.black87,
        foregroundColor: Colors.white,
        elevation: 2,
        actions: [
          IconButton(icon: const Icon(Icons.refresh), onPressed: () => setState(() {})),
        ],
      ),
      body: FutureBuilder<List<StreamModel>>(
        future: StreamService.fetchStreams(),
        builder: (context, snapshot) {
          if (snapshot.connectionState == ConnectionState.waiting) return const Center(child: CircularProgressIndicator());
          if (snapshot.hasError) {
            return Center(
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Icon(Icons.error_outline, size: 64, color: Colors.red[300]),
                  const SizedBox(height: 16),
                  Text('Erro ao carregar lives: ${snapshot.error}'),
                  ElevatedButton.icon(
                    onPressed: () => setState(() {}),
                    icon: const Icon(Icons.refresh),
                    label: const Text('Tentar novamente'),
                  ),
                ],
              ),
            );
          }

          final streams = snapshot.data ?? [];
          if (streams.isEmpty) {
            return const Center(child: Text('Nenhuma live ativa'));
          }

          return RefreshIndicator(
            onRefresh: () async => setState(() {}),
            child: GridView.builder(
              padding: const EdgeInsets.all(16),
              gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
                crossAxisCount: 2,
                childAspectRatio: 0.8,
                crossAxisSpacing: 12,
                mainAxisSpacing: 12,
              ),
              itemCount: streams.length,
              itemBuilder: (context, index) => LiveCard(
                stream: streams[index],
                onTap: () => Navigator.push(
                  context,
                  MaterialPageRoute(builder: (_) => LiveRoomScreen(streamId: streams[index].id)),
                ),
              ),
            ),
          );
        },
      ),
    );
  }
}
```

## 7. lib/screens/live_room_screen.dart (Live + Chat)
```
import 'package:flutter/material.dart';
import 'package:livekit_client/livekit_client.dart';
import '../services/stream_service.dart';
import '../widgets/chat_overlay.dart';

class LiveRoomScreen extends StatefulWidget {
  final String streamId;
  const LiveRoomScreen({super.key, required this.streamId});

  @override
  State<LiveRoomScreen> createState() => _LiveRoomScreenState();
}

class _LiveRoomScreenState extends State<LiveRoomScreen> {
  Room? room;
  bool _isConnected = false;
  int _viewersCount = 0;
  VideoRenderer _renderer = VideoRenderer();

  @override
  void initState() {
    super.initState();
    _renderer.initialize();
    _connectToLive();
  }

  Future<void> _connectToLive() async {
    try {
      final tokenData = await StreamService.getLivekitToken(widget.streamId);

      room = await Room.connect(
        url: 'wss://livestreamapp-ghcjo92o.livekit.cloud',
        token: tokenData['token'],
      );

      room!.on<RemoteTrackAddedEvent>((event) => setState(() {}));
      room!.on<ParticipantEvent>((_) => setState(() => _viewersCount++));

      setState(() => _isConnected = true);
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('$e')));
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.black,
      appBar: AppBar(
        backgroundColor: Colors.transparent,
        leading: IconButton(icon: const Icon(Icons.arrow_back, color: Colors.white), onPressed: () => Navigator.pop(context)),
        actions: [
          Container(
            padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 4),
            margin: const EdgeInsets.only(right: 16),
            decoration: BoxDecoration(color: Colors.red, borderRadius: BorderRadius.circular(20)),
            child: Text('$_viewersCount viewers', style: const TextStyle(color: Colors.white, fontSize: 12)),
          ),
        ],
      ),
      body: Stack(
        children: [
          // Video
          if (_isConnected) VideoView(_renderer, objectFit: VideoViewObjectFit.cover)
          else const Center(child: CircularProgressIndicator(color: Colors.white)),
          // Chat
          Positioned(bottom: 0, left: 0, right: 0, child: ChatOverlay(streamId: widget.streamId)),
        ],
      ),
    );
  }

  @override
  void dispose() {
    room?.disconnect();
    super.dispose();
  }
}
```
